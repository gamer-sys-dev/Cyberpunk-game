   async function fetchWithRetry(url, options, retries = 3) {
       const delays = [1000, 2000, 4000];
       for (let i = 0; i < retries; i++) {
           try {
               const response = await fetch(url, options);
               if (!response.ok) throw new Error(`HTTP error ${response.status}`);
               return await response.json();
           } catch (error) {
               if (i === retries - 1) throw error;
               await new Promise(res => setTimeout(res, delays[i]));
           }
       }
   }

   function wrapText(text, x, y, maxWidth, lineHeight) {
       if (!text) return;
       const paragraphs = String(text).split('\n'); let currentY = y;
       for (let p = 0; p < paragraphs.length; p++) {
           const words = paragraphs[p].split(' '); let line = '';
           for(let n = 0; n < words.length; n++) {
               const testLine = line + words[n] + ' ';
               if (ctx.measureText(testLine).width > maxWidth && n > 0) { ctx.fillText(line, x, currentY); line = words[n] + ' '; currentY += lineHeight; }
               else { line = testLine; }
           }
           ctx.fillText(line, x, currentY); currentY += lineHeight + (lineHeight * 0.5);
       }
   }

           if(Math.random() > 0.9) item.multStr = 2.0;
       } else if (slot === 'armor') { item.name = ["Titanium", "Aegis", "Kevlar"][Math.floor(Math.random()*3)] + " Chassis"; item.multHp = 1.0 + (Math.floor(Math.random() * 10) * 0.1);
       } else if (slot === 'shoes') { item.name = ["Hermes", "Sonic", "Aero"][Math.floor(Math.random()*3)] + " Treads"; item.multSpd = 1.0 + (Math.floor(Math.random() * 5) * 0.2);
       } else if (slot === 'accessory') {
           item.name = ["Logic", "Data", "Matrix"][Math.floor(Math.random()*3)] + " Core";
           const roll = Math.random();
           if(roll > 0.6) item.multXp = [2, 3, 10][Math.floor(Math.random()*3)]; else if(roll > 0.3) { item.multStr = 2.0; item.multTech = 2.0; } else { item.multHp = 2.0; item.multSpd = 2.0; }
       }
       return item;
   }

   function formatItemStats(item) {
       if (!item) return "NONE"; let stats = [];
       if (item.bonusStr > 0) stats.push(`+${item.bonusStr} STR`); if (item.bonusTech > 0) stats.push(`+${item.bonusTech} TECH`);
       if (item.multHp > 1) stats.push(`x${item.multHp.toFixed(1)} HP`); if (item.multSpd > 1) stats.push(`x${item.multSpd.toFixed(1)} SPD`);
       if (item.multStr > 1) stats.push(`x${item.multStr.toFixed(1)} STR`); if (item.multTech > 1) stats.push(`x${item.multTech.toFixed(1)} TECH`);
       if (item.multXp > 1) stats.push(`x${item.multXp} XP`);
       return stats.join(", ");
   }

   function spawnEntity() {
       if (Math.random() < 0.035) {
           const rand = Math.random();
           if (currentStage === 1 && rand < 0.25) entities.push({ globalX: cameraX + 900, y: 400, w: 20, h: 50, type: 'CURSE', active: true });
           else if (currentStage === 1 && rand < 0.40) entities.push({ globalX: cameraX + 900, y: 280, w: 30, h: 30, type: 'BLESSING', active: true });
           else if (rand < 0.70) entities.push({ globalX: cameraX + 900, y: 435, w: 20, h: 20, type: 'XP', active: true });
           else entities.push({ globalX: cameraX + 900, y: 435, w: 20, h: 20, type: 'HP', active: true });
       }
   }

   function executeCombatTrigger(type) {
       gameState = 'LOADING_ENCOUNTER'; encounterType = type; player.distanceTraveled = 0; entities = [];
       
       fetchDynamicEnemy(player.level, type).then(data => {
           let mult = type === 'BOSS' ? 4 : (type === 'TOURNAMENT' ? 2 : 1);
           if (data) {
               let safeMaxHp = parseInt(data.maxHp) || 40; let scaledHp = Math.max(safeMaxHp, 40 * player.level * mult);
               enemy = { name: data.name || "Construct", hp: scaledHp, maxHp: scaledHp, atb: 0, maxAtb: 100, speedStat: parseInt(data.speedStat) || 20, color: data.colorHex || '#ff0055', entranceQuote: data.entranceQuote || "Engaging.", attackQuote: data.attackQuote || "Attacking." };
           } else {
               let sHp = fallbackEnemy.maxHp * player.level * mult; enemy = { ...fallbackEnemy, name: type + ' Construct', hp: sHp, maxHp: sHp };
           }
           
           if (type === 'PET') { pendingPet = petList[Math.floor(Math.random() * petList.length)]; enemy.name = pendingPet.name; }
           
           gameState = 'COMBAT'; player.atb = 0; combat.turnState = 'FILLING'; combat.freezeTimer = 0;
           combat.message = enemy.entranceQuote; combat.messageTimer = 3; combat.menuIndex = 0; keys['Enter'] = false;
       });
   }

   function processLevelUps() {
       let leveled = false;
       while (player.xp >= player.nextLevelXp) {
           player.level++; player.xp -= player.nextLevelXp; player.nextLevelXp = Math.floor(player.nextLevelXp * 1.5); player.skillPoints += 3; leveled = true;
       }
       if (leveled) { player.updateDerivedStats(); player.hp = player.maxHp; player.mp = player.maxMp; postCombatMessage += `\nLEVEL UP! Reached Level ${player.level}.`; }
   }

   function handlePostCombatRewards() {
       player.activeCurse = null; player.activeBlessing = null; player.updateDerivedStats();
       let baseReward = encounterType === 'BOSS' ? 200 : (encounterType === 'TOURNAMENT' ? 80 : 40);
       let earnedXp = Math.floor((baseReward * player.level) * player.xpMultiplier);
       player.xp += earnedXp; postCombatMessage = `Gained ${earnedXp} XP.`;
       processLevelUps();

       if (encounterType === 'STANDARD' && currentStage === 1) {
           stageEnemiesDefeated++;
           if (stageEnemiesDefeated >= 5) { entities.push({ globalX: cameraX + 400, y: 435, w: 30, h: 20, type: 'STAGE_CHEST', active: true }); stageEnemiesDefeated = 0; }
       }
       else if (encounterType === 'PET') { player.pet = pendingPet; player.updateDerivedStats(); postCombatMessage += `\nAcquired Companion: ${pendingPet.name}!`; }
       else if (encounterType === 'TOURNAMENT') { if (Math.random() > 0.5) player.inventory.push(generateChestLoot()); }
   }

   function finalizePlayerTurn() {
       player.atb = 0; combat.messageTimer = 3.5;
       if (enemy.hp <= 0) {
           enemy.hp = 0; combat.turnState = 'END'; handlePostCombatRewards();
           setTimeout(() => {
               if (gameState === 'COMBAT') {
                   gameState = 'LOADING_STORY';
                   fetchNextStoryBeat(enemy.name, encounterType).then(newStory => {
                       storyText = `${postCombatMessage}\n\n${newStory}`;
                       if (encounterType === 'BOSS') {
                           if (player.amulet.length < 4) { gameState = 'AMULET_SELECTION'; amuletMenu.index = 0; postAmuletState = 'STORY_BEAT'; currentStage = 1; keys['Enter'] = false; return; }
                           currentStage = 1;
                       }
                       else if (encounterType === 'PET') { currentStage = 3; tournamentRound = 1; storyText += "\n\nEntering the Underground Tournament Arena..."; }
                       else if (encounterType === 'TOURNAMENT') {
                           tournamentRound++;
                           if (tournamentRound > 3) { currentStage = 4; storyText += "\n\nTournament cleared! Approaching Omega Construct territory..."; }
                           else { gameState = 'TOURNAMENT_LOBBY'; keys['Enter'] = false; return; }
                       }
                       gameState = 'STORY_BEAT'; keys['Enter'] = false;
                   });
               }
           }, 2000);
       } else { combat.turnState = 'FILLING'; }
   }

   function executePlayerAction() {
       const option = combat.options[combat.menuIndex];
       if (option === 'Attack') {
           const dmg = Math.floor(Math.random() * 5) + player.baseDamage;
           enemy.hp -= dmg; createParticle(600, 300, dmg, '#fff'); combat.message = `Physical strike: ${dmg} damage!`; finalizePlayerTurn();
       } else if (option.includes('Magic')) {
           if (player.mp >= 10) {
               player.mp -= 10; const dmg = Math.floor(Math.random() * 10) + player.magicDamage;
               enemy.hp -= dmg; createParticle(600, 300, dmg, '#00ffcc'); combat.message = `Neon Blast: ${dmg} tech damage!`; finalizePlayerTurn();
           } else { combat.message = "Not enough MP!"; }
       } else if (option.includes('AI Hack')) {
           if (player.mp >= 20) {
               player.mp -= 20; combat.turnState = 'AWAITING_HACK'; combat.message = "✨ Calling Neural Network... ✨";
               fetchAIHack(enemy.name).then(hack => {
                   const totalDamage = player.magicDamage + (parseInt(hack.damage) || 50);
                   enemy.hp -= totalDamage; createParticle(600, 300, totalDamage, '#ff00cc');
                   combat.message = String(hack.description || "Neural link established."); finalizePlayerTurn();
               });
           } else { combat.message = "Not enough MP!"; }
       }
   }

   function drawParallaxBackground(ctx, camX, time, stage) {
       let bgBase = '#020205', sky = '#0a0f1a', mid = '#111b2b', lines = '#00ffcc', drop = 'rgba(0, 255, 204, 0.4)';
       if (stage === 2) { bgBase = '#1a0500'; sky = '#331100'; mid = '#552200'; lines = '#ffaa00'; drop = 'rgba(255, 150, 0, 0.4)'; }
       else if (stage === 3) { bgBase = '#050011'; sky = '#110022'; mid = '#220044'; lines = '#cc00ff'; }
       else if (stage === 4) { bgBase = '#220000'; sky = '#440000'; mid = '#110000'; lines = '#ff0000'; }

       ctx.fillStyle = bgBase; ctx.fillRect(0, 0, canvas.width, canvas.height);
       const bgGrad = ctx.createRadialGradient(400, 300, 100, 400, 300, 600); bgGrad.addColorStop(0, sky); bgGrad.addColorStop(1, bgBase);
       ctx.fillStyle = bgGrad; ctx.fillRect(0, 0, canvas.width, canvas.height);

       if (stage === 3) {
           ctx.fillStyle = 'rgba(255, 255, 255, 0.05)';
           ctx.beginPath(); ctx.moveTo(200, 0); ctx.lineTo(100, 450); ctx.lineTo(300, 450); ctx.fill();
           ctx.beginPath(); ctx.moveTo(600, 0); ctx.lineTo(500, 450); ctx.lineTo(700, 450); ctx.fill();
           ctx.strokeStyle = lines; ctx.lineWidth = 5; ctx.beginPath(); ctx.moveTo(0, 450); ctx.lineTo(canvas.width, 450); ctx.stroke();
           return;
       }

       ctx.fillStyle = sky;
       for (let i = 0; i < 20; i++) { let bX = (i * 100) - (camX * 0.1) % 100; ctx.fillRect(bX, 150 + Math.sin(i)*60, 90, 450); if(i%3===0 && stage !== 4) { ctx.fillStyle = lines; ctx.fillRect(bX+20, 200 + Math.sin(i)*60, 5, 5); ctx.fillStyle = sky; } }
       ctx.fillStyle = mid;
       for (let i = 0; i < 15; i++) { let bX = (i * 150) - (camX * 0.3) % 150; ctx.fillRect(bX, 250 + Math.cos(i)*80, 140, 350); }
       ctx.strokeStyle = lines; ctx.lineWidth = 3; ctx.beginPath(); ctx.moveTo(0, 450); ctx.lineTo(canvas.width, 450); ctx.stroke();
       ctx.strokeStyle = `rgba(100, 100, 100, 0.3)`;
       for (let i = 0; i < 30; i++) { let lineX = (i * 40) - (camX * 1.0) % 40; ctx.beginPath(); ctx.moveTo(lineX, 450); ctx.lineTo(lineX - 250, 600); ctx.stroke(); }
       if (stage !== 4) { ctx.strokeStyle = drop; ctx.lineWidth = 1; ctx.beginPath(); for (const d of rain) { ctx.moveTo(d.x, d.y); ctx.lineTo(d.x - (d.length * 0.2), d.y + d.length); } ctx.stroke(); }
   }

   function drawPlayerCharacter(ctx, x, y, time, isMoving, isCombat) {
       let bob = isMoving && !player.isJumping ? Math.sin(time * 15) * 8 : 0;
       if (!isMoving && !player.isJumping) bob = Math.sin(time * 3) * 3;
       const lean = isMoving && !player.isJumping ? 15 : 0;
       
       ctx.save();
       let glitchX = 0, glitchY = 0;
       if (isCombat && player.activeCurse === 'GLITCH_DANCE') { glitchX = (Math.random() - 0.5) * 15; glitchY = (Math.random() - 0.5) * 15; }

       ctx.translate(x + glitchX, y + bob + glitchY);
       if (isCombat && player.activeCurse === 'MINIATURIZED') ctx.scale(0.5, 0.5);
       if (isMoving && !player.isJumping) ctx.rotate(lean * Math.PI / 180);

       ctx.strokeStyle = player.color; ctx.lineWidth = 2; ctx.stroke();
       ctx.fillStyle = '#005555'; ctx.fillRect(-5, -40, 15, 10);

   }

       if (!enemyData) return;
       const hover = Math.sin(time * 4) * 12; const scale = enemyData.isBoss ? 2.5 : (encounterType === 'TOURNAMENT' ? 1.5 : 1.0);

       ctx.restore();
   }

   // --- MAIN LOOP UPDATE & DRAW ---
   function update(deltaTime) {
       globalTime += deltaTime;
       for (const drop of rain) { drop.y += drop.speed * deltaTime; drop.x -= (drop.speed * 0.2) * deltaTime; if (drop.y > 600) { drop.y = -20; drop.x = Math.random() * 1000; } }

       if (gameState === 'CHARACTER_CREATION') return;
       
       // Use keys tied to Virtual Pad
       if (gameState === 'AMULET_SELECTION') {
           if (keys['ArrowDown']) { amuletMenu.index = Math.min(amuletMenu.index + 1, amuletMenu.options.length - 1); keys['ArrowDown'] = false; }
           if (keys['ArrowUp']) { amuletMenu.index = Math.max(amuletMenu.index - 1, 0); keys['ArrowUp'] = false; }
           if (keys['Enter']) { player.amulet.push(amuletMenu.options[amuletMenu.index].id); player.updateDerivedStats(); gameState = postAmuletState; keys['Enter'] = false; }
       }
       else if (gameState === 'STORY_INTRO' || gameState === 'STORY_BEAT') {
           if (keys['Enter']) { gameState = currentStage === 3 ? 'TOURNAMENT_LOBBY' : (currentStage === 4 ? 'BOSS_PROMPT' : 'EXPLORE'); keys['Enter'] = false; }
       }
       else if (gameState === 'TOURNAMENT_LOBBY') {
           if (keys['Enter']) { executeCombatTrigger('TOURNAMENT'); keys['Enter'] = false; }
           if (keys['KeyU']) { gameState = 'UPGRADE_MENU'; upgradeMenu.index = 0; keys['KeyU'] = false; }
           if (keys['KeyI']) { gameState = 'INVENTORY_MENU'; inventoryMenu.index = 0; keys['KeyI'] = false; }
           if (keys['KeyR']) {
               let spent = (player.stats.str - 5) + (player.stats.vit - 5) + (player.stats.tech - 5) + (player.stats.spd - 5);
               if (spent > 0) { player.skillPoints += spent; player.stats = { str: 5, vit: 5, tech: 5, spd: 5 }; player.updateDerivedStats(); }
               keys['KeyR'] = false;
           }
       }
       else if (gameState === 'EXPLORE') {
           player.isMoving = false; let dx = 0;
           if (keys['ArrowRight']) dx = 1; if (keys['ArrowLeft']) dx = -1;
           
           if (keys['Space'] && !player.isJumping) { player.vy = -600; player.isJumping = true; keys['Space'] = false; /* Debounce jump slightly */ }
           
           if (player.isJumping || player.y < player.baseY) {
               player.vy += 1500 * deltaTime; player.y += player.vy * deltaTime;
               if (player.y >= player.baseY) { player.y = player.baseY; player.vy = 0; player.isJumping = false; }
           }
           
           if (keys['KeyU']) { gameState = 'UPGRADE_MENU'; upgradeMenu.index = 0; keys['KeyU'] = false; }
           if (keys['KeyI']) { gameState = 'INVENTORY_MENU'; inventoryMenu.index = 0; keys['KeyI'] = false; }

           if (dx !== 0) {
               player.isMoving = true; cameraX += dx * player.speed * deltaTime; player.distanceTraveled += Math.abs(dx * player.speed * deltaTime);
               spawnEntity();
               const pRect = { x: cameraX + 135, y: player.y - 60, w: 30, h: 60 };
               for (let i = entities.length - 1; i >= 0; i--) {
                   const item = entities[i]; if (!item.active) continue;
                   const eRect = { x: item.globalX - (item.w/2), y: item.y - (item.h/2), w: item.w, h: item.h };
                   if (item.type === 'CURSE') { eRect.y = item.y; eRect.x = item.globalX - 10; }
                   if (item.type === 'CHEST' || item.type === 'STAGE_CHEST') { eRect.y = item.y - 20; }
                   if (pRect.x < eRect.x + eRect.w && pRect.x + pRect.w > eRect.x && pRect.y < eRect.y + eRect.h && pRect.y + pRect.h > eRect.y) {
                       item.active = false;
                       if (item.type === 'CURSE') { player.activeCurse = cursesList[Math.floor(Math.random() * cursesList.length)]; createParticle(150, player.y - 50, `CURSED!`, "#f00"); }
                       else if (item.type === 'BLESSING') { player.activeBlessing = blessingsList[Math.floor(Math.random() * blessingsList.length)]; player.updateDerivedStats(); createParticle(150, player.y - 50, `BLESSED!`, "#fd0"); }
                       else if (item.type === 'STAGE_CHEST') { currentLootDisplay = generateChestLoot(); gameState = 'CHEST_REWARD'; keys['Enter'] = false; currentStage = 2; storyText = "Leaving the Neon Slums...\n\nEntering the Global Outskirts.\nSearch this desert zone to find a Cyber-Pet companion."; }
                       else if (item.type === 'CHEST') { currentLootDisplay = generateChestLoot(); gameState = 'CHEST_REWARD'; keys['Enter'] = false; }
                       else if (item.type === 'HP') { player.hp = Math.min(player.maxHp, player.hp + 30); createParticle(150, player.y - 30, "+30 HP", "#0f0"); }
                       else { player.xp += 15 * player.level * player.xpMultiplier; processLevelUps(); createParticle(150, player.y - 30, "+XP", "#ff0"); }
                   }
               }
               entities = entities.filter(item => item.active && item.globalX > cameraX - 500 && item.globalX < cameraX + 1500);
               if (player.distanceTraveled > 800 && Math.random() < 0.02) { keys['ArrowLeft']=false; keys['ArrowRight']=false; triggerEncounterCheck(); }
           }
       }
       else if (gameState === 'BOSS_PROMPT') {
           if (keys['Y']) { executeCombatTrigger('BOSS'); keys['Y']=false; }
           else if (keys['N']) { currentStage = 1; stageEnemiesDefeated = 0; storyText = "Retreating to the Neon Slums to train..."; gameState = 'STORY_BEAT'; keys['N'] = false; }
       }
       else if (gameState === 'CHEST_REWARD') {
           if (keys['Enter']) { player.inventory.push(currentLootDisplay); if (currentStage === 2 && encounterType === 'STANDARD') gameState = 'STORY_BEAT'; else gameState = 'EXPLORE'; keys['Enter'] = false; }
       }
       else if (gameState === 'UPGRADE_MENU') {
           if (keys['ArrowDown']) { upgradeMenu.index = Math.min(upgradeMenu.index + 1, upgradeMenu.options.length - 1); keys['ArrowDown'] = false; }
           if (keys['ArrowUp']) { upgradeMenu.index = Math.max(upgradeMenu.index - 1, 0); keys['ArrowUp'] = false; }
           if (keys['Enter']) {
               const sel = upgradeMenu.options[upgradeMenu.index];
               if (sel === 'exit') { gameState = currentStage === 3 ? 'TOURNAMENT_LOBBY' : 'EXPLORE'; }
               else if (player.skillPoints > 0) { player.stats[sel]++; player.skillPoints--; player.updateDerivedStats(); }
               keys['Enter'] = false;
           }
       }
       else if (gameState === 'INVENTORY_MENU') {
           const maxIdx = Math.max(0, player.inventory.length);
           if (keys['ArrowDown']) { inventoryMenu.index = Math.min(inventoryMenu.index + 1, maxIdx); keys['ArrowDown'] = false; }
           if (keys['ArrowUp']) { inventoryMenu.index = Math.max(inventoryMenu.index - 1, 0); keys['ArrowUp'] = false; }
           if (keys['Enter']) {
               if (inventoryMenu.index === player.inventory.length) { gameState = currentStage === 3 ? 'TOURNAMENT_LOBBY' : 'EXPLORE'; }
               else if (player.inventory.length > 0) { const selectedItem = player.inventory[inventoryMenu.index]; player.gear[selectedItem.slot] = selectedItem; player.updateDerivedStats(); }
               keys['Enter'] = false;
           }
       }
       else if (gameState === 'COMBAT') {
           for (let i = combat.particles.length - 1; i >= 0; i--) { const p = combat.particles[i]; p.y += p.velocityY * deltaTime; p.life -= deltaTime; if (p.life <= 0) combat.particles.splice(i, 1); }
           if (combat.turnState === 'FILLING') {
               if (player.activeCurse === 'SYSTEM_FREEZE' && Math.random() < 0.01) combat.freezeTimer = 1.0;
               if (combat.freezeTimer > 0) { combat.freezeTimer -= deltaTime; combat.message = "Curse: SYSTEM FROZEN!"; } else { player.atb += player.speedStat * deltaTime; }
               enemy.atb += enemy.speedStat * deltaTime;
               if (player.atb >= player.maxAtb) { player.atb = player.maxAtb; combat.turnState = 'PLAYER_ACTION'; keys['Enter'] = false;
               } else if (enemy.atb >= enemy.maxAtb) {
                   enemy.atb = enemy.maxAtb;
                   const baseEDmg = encounterType === 'BOSS' ? 20 : (encounterType === 'TOURNAMENT' ? 10 : 5); const dmg = Math.floor(Math.random() * 10) + (baseEDmg * player.level);
                   player.hp -= dmg; createParticle(150, 300, dmg, '#ff0055'); combat.message = `${enemy.attackQuote} (${dmg} dmg!)`; enemy.atb = 0; combat.messageTimer = 2.0;
                   if (player.hp <= 0) { player.hp = 0; gameState = 'GAME_OVER'; }
               }
           } else if (combat.turnState === 'PLAYER_ACTION') {
               if (keys['ArrowDown']) { combat.menuIndex = Math.min(combat.menuIndex+1, 2); keys['ArrowDown'] = false; }
               if (keys['ArrowUp']) { combat.menuIndex = Math.max(combat.menuIndex-1, 0); keys['ArrowUp'] = false; }
               if (keys['Enter']) { executePlayerAction(); keys['Enter'] = false; }
           }
           if (combat.messageTimer > 0) combat.messageTimer -= deltaTime;
       }
   }

   function draw() {

           ctx.fillStyle = '#ff00cc'; ctx.font = 'bold 28px Courier New'; ctx.textAlign = 'center'; ctx.fillText(`CHOOSE ANOMALY GEM (SLOT ${player.amulet.length + 1}/4)`, canvas.width/2, 100);

               ctx.fillStyle = i === amuletMenu.index ? '#00ffcc' : '#aaa'; ctx.fillText(`${i === amuletMenu.index ? '>> ' : '   '} ${amuletMenu.options[i].label}`, 150, startY + (i * 40));

           ctx.fillStyle = '#fff'; ctx.font = '16px Courier New'; ctx.fillText(`OPERATIVE: ${player.name} [STAGE ${currentStage}]`, 20, 30);
           ctx.fillText(`LVL: ${player.level} | XP: ${player.xp}/${player.nextLevelXp} | HP: ${player.hp}/${player.maxHp} | MP: ${player.mp}/${player.maxMp}`, 20, 50);

           ctx.fillStyle = '#cc00ff'; ctx.font = 'bold 30px Courier New'; ctx.textAlign = 'center'; ctx.fillText("THE UNDERGROUND TOURNAMENT", canvas.width/2, 160);
           ctx.fillStyle = '#fff'; ctx.font = '20px Courier New'; ctx.fillText(`ROUND ${tournamentRound} of 3`, canvas.width/2, 220);

           ctx.font = '20px Courier New'; ctx.fillText("Are you prepared to engage?", canvas.width/2, 300); ctx.textAlign = 'left';
       }
       else if (gameState === 'CHEST_REWARD') {
           ctx.fillStyle = 'rgba(0, 10, 20, 0.9)'; ctx.fillRect(100, 150, 600, 300); ctx.strokeStyle = '#ff00cc'; ctx.lineWidth = 4; ctx.strokeRect(100, 150, 600, 300);
           ctx.fillStyle = '#ff00cc'; ctx.font = 'bold 24px Courier New'; ctx.textAlign = 'center'; ctx.fillText("RARE CACHE OPENED!", canvas.width/2, 200);
           ctx.fillStyle = '#fff'; ctx.font = '20px Courier New'; ctx.fillText(`Looted: ${currentLootDisplay.name}`, canvas.width/2, 260);
           ctx.fillStyle = '#00ffcc'; ctx.font = '16px Courier New'; ctx.fillText(`Effects: ${formatItemStats(currentLootDisplay)}`, canvas.width/2, 300); ctx.textAlign = 'left';

           } else { ctx.fillStyle = '#666'; ctx.fillText("Awaiting Turn...", 410, 490); }

       }
       else if (gameState === 'GAME_OVER') {
           ctx.fillStyle = 'rgba(255, 0, 85, 0.3)'; ctx.fillRect(0, 0, canvas.width, canvas.height); ctx.fillStyle = '#fff'; ctx.font = '40px Courier New'; ctx.textAlign = 'center'; ctx.fillText("SYSTEM FAILURE", canvas.width/2, canvas.height/2);
       }

       ctx.filter = 'none';

       // Draw Virtual Joypad (If not Character Creation or Game Over)
       if (gameState !== 'CHARACTER_CREATION' && gameState !== 'GAME_OVER') {
           vPad.draw(ctx, gameState);
       }
   }

   function gameLoop(timestamp) {
       if (!lastTime) lastTime = timestamp;
       let deltaTime = (timestamp - lastTime) / 1000; lastTime = timestamp;
       if (deltaTime > 0.1) deltaTime = 0.1;
       update(deltaTime); draw(); requestAnimationFrame(gameLoop);
   }
   requestAnimationFrame(gameLoop);
